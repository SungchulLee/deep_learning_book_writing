    print("""
    온전한 그림: 베이즈 추론에서 확산까지
    ==========================================================

    개념의 온 여정을 짚어 보자.

    1. 베이즈 미룸(01_Bayesian_Inference 단원):
       -----------------------------------------------------
       Problem: p(θ|D) = p(D|θ)p(θ) / p(D)
       어려움: p(D) = ∫ p(D|θ)p(θ) dθ을 셈할 수 없다

       필요한 풀이: 고르게 맞추지 않은 분포에서 뽑기!

    2. 점수 함수(1단원):
       -----------------------------
       핵심 눈썰미: s(x) = ∇_x log p(x)

       성질:
       ✓ 확률 높은 쪽을 가리킨다
       ✓ 고르게 맞추기가 필요 없다!
       ✓ 봉우리에서 0이다

       그런데 자료만으로 어떻게 셈하는가?

    3. 점수 맞추기(2단원):
       ----------------------------
       표본에서 s_θ(x) ≈ ∇_x log p_data(x)을 배운다

       핵심 재주: 잡음 지우기 점수 맞추기(DSM)
       - Add noise: x̃ = x + σε
       - 헤아리는 법을 배운다: s_θ(x̃) ≈ -ε/σ

       이음: 잡음 없애기가 곧 베이즈 추론이다!
       뒷분포 p(x|x̃) → 점수가 잡음 지우는 법을 알려 준다

       -----------------------------

       Training = DSM at multiple times
       Sampling = Reverse diffusion

                                     Relation: s(x,t) = -ε/√(1-ᾱ_t)

    잡음 없애는 점수 맞추기      ↔  잡음 헤아리기 손실
                                     Both: ||ε - ε_θ||²

    랑주뱅 움직임             ↔  뒤 확산
                                     같은 뽑기 절차!

    식힘 랑주뱅            ↔  여러 걸음 잡음 없애기
                                     차츰 잡음을 없앤다

    흩어짐 터짐 확률 미분 방정식                       ↔  잡음 조건 모델
    흩어짐 지키기 확률 미분 방정식                       ↔  DDPM 적기

    확률 흐름 상미분 방정식         ↔  DDIM 뽑기
                                     정해진 만들어 내기

    온전한 DDPM 꼴:

    class DDPM:
        """
        잡음 없애는 확산 확률 모델

        이것이 여태 배운 모든 것을 하나로 만든다!
        """
        def __init__(self, n_timesteps=1000, beta_start=0.0001, beta_end=0.02):
            self.n_timesteps = n_timesteps

            # 선형 차례표(다른 것도 쓸 수 있다)
            self.betas = torch.linspace(beta_start, beta_end, n_timesteps)
            self.alphas = 1 - self.betas
            self.alphas_cumprod = torch.cumprod(self.alphas, dim=0)

            # 쓸모 있는 양을 미리 셈한다
            self.sqrt_alphas_cumprod = torch.sqrt(self.alphas_cumprod)
            self.sqrt_one_minus_alphas_cumprod = torch.sqrt(1 - self.alphas_cumprod)

        def q_sample(self, x_0, t, noise=None):
            """
            앞 과정: 잡음 더하기

            q(x_t|x_0) = N(x_t; √ᾱ_t x_0, (1-ᾱ_t)I)

            This is: x_t = √ᾱ_t x_0 + √(1-ᾱ_t) ε

            점수 맞추기와의 이음:
            - 이것이 잡음 없애는 점수 맞추기의 "잡음 더하기"이다!
            - t가 다르면 잡음 층 σ도 다르다
            """
            if noise is None:
                noise = torch.randn_like(x_0)

            sqrt_alpha_bar = self.sqrt_alphas_cumprod[t]
            sqrt_one_minus_alpha_bar = self.sqrt_one_minus_alphas_cumprod[t]

            # 퍼뜨리기에 맞게 꼴을 바꾼다
            while len(sqrt_alpha_bar.shape) < len(x_0.shape):
                sqrt_alpha_bar = sqrt_alpha_bar.unsqueeze(-1)
                sqrt_one_minus_alpha_bar = sqrt_one_minus_alpha_bar.unsqueeze(-1)

            return sqrt_alpha_bar * x_0 + sqrt_one_minus_alpha_bar * noise

        def training_loss(self, model, x_0):
            """
            익히기 목표

            L = 𝔼_t 𝔼_ε ||ε - ε_θ(x_t, t)||²

            DSM과의 이음:
            - 이것이 바로 잡음 없애는 점수 맞추기이다!
            - DSM에서 t가 다르면 σ도 다르다
            - ε을 헤아리는 것은 점수를 헤아리는 것과 같다

            이끌어 내기:
            Score s(x_t,t) = ∇log p(x_t|x_0)
                           = -ε / √(1-ᾱ_t)

            So: ε = -√(1-ᾱ_t) * score
            """
            # 아무 때 걸음
            batch_size = x_0.shape[0]
            t = torch.randint(0, self.n_timesteps, (batch_size,))

            # 잡음을 더한다(앞 과정)
            noise = torch.randn_like(x_0)
            x_t = self.q_sample(x_0, t, noise)

            # 잡음을 헤아린다
            predicted_noise = model(x_t, t)

            # 손실
            return torch.mean((noise - predicted_noise) ** 2)

        def p_sample(self, model, x_t, t):
            """
            뒤 과정: 잡음 없애기 걸음 하나

            p_θ(x_{t-1}|x_t) = N(x_{t-1}; μ_θ(x_t,t), σ_t²I)

            여기서 각 기호는 다음과 같다.
            μ_θ = (1/√α_t)[x_t - (β_t/√(1-ᾱ_t))ε_θ(x_t,t)]

            랑주뱅과의 이음:
            - 이것이 식힘 랑주뱅의 한 걸음이다!
            - ε_θ이 점수 방향을 준다
            - μ_θ이 랑주뱅 고침이다
            """
            # 잡음을 헤아린다
            predicted_noise = model(x_t, t)

            # 계수를 뽑는다
            alpha_t = self.alphas[t]
            alpha_bar_t = self.alphas_cumprod[t]
            beta_t = self.betas[t]

            # 퍼뜨리기에 맞게 꼴을 바꾼다
            while len(alpha_t.shape) < len(x_t.shape):
                alpha_t = alpha_t.unsqueeze(-1)
                alpha_bar_t = alpha_bar_t.unsqueeze(-1)
                beta_t = beta_t.unsqueeze(-1)

            # 뒤 분포의 평균
            mean = (1 / torch.sqrt(alpha_t)) * (
                x_t - (beta_t / torch.sqrt(1 - alpha_bar_t)) * predicted_noise
            )

            # 잡음을 더한다(t=0만 빼고)
            if t[0] > 0:
                noise = torch.randn_like(x_t)
                variance = beta_t
                return mean + torch.sqrt(variance) * noise
            else:
                return mean

        def sample(self, model, shape):
            """
            온전한 뽑기: 잡음에서 만들어 내기

            This is:
            1. 식힘 랑주뱅 움직임
            2. 뒤 확률 미분 방정식과 상미분 방정식
            3. 잡음 없애는 확산

            모두 같은 것이다!
            """
            # 순수 잡음에서 시작한다
            x = torch.randn(shape)

            # 뒤 확산
            for t in reversed(range(self.n_timesteps)):
                t_batch = torch.ones(shape[0], dtype=torch.long) * t
                x = self.p_sample(model, x, t_batch)

            return x

## 2. 정방향 노이즈와 주변 스코어

클래스는 βt를 선형으로 만들고 αt=1−βt와 누적곱 ᾱt를 저장한다. q_sample은 다음 관측을 직접 생성한다.

$$x_t=\sqrt{\bar\alpha_t}x_0+\sqrt{1-\bar\alpha_t}\,\epsilon,\qquad \epsilon\sim N(0,I)$$

배치별 t가 길이 B이면 스케줄 계수를 뒤쪽 축에 확장해 B×d나 B×C×H×W 데이터와 곱한다. 반환 모양은 입력과 같다. 코드 인덱스 0은 첫 번째 β가 이미 적용된 노이즈 단계이므로 노이즈가 전혀 없는 원본 시점과 같지 않다.

조건부 관측 스코어는 −ε/√(1−ᾱt)다. 노이즈 MSE의 모집단 최적 예측은 관측 xₜ에 대한 ε의 조건부 평균이므로, 해당 예측과 주변 스코어는 다음처럼 연결된다.

$$s_\theta(x_t,t)=-\frac{\epsilon_\theta(x_t,t)}{\sqrt{1-\bar\alpha_t}}$$

정확한 모집단 관계와 유한 네트워크의 근사를 구분한다. 단일 표본의 실제 ε를 관측만으로 언제나 정확히 복원할 수 있다는 뜻은 아니다. 베이즈와의 연결도 깨끗한 x₀에 대한 사후 **평균**과 노이즈 주변 스코어의 관계이며 서로 다른 좌표의 사후 스코어를 동일시하는 것이 아니다.

## 3. 단순 노이즈 손실과 변분 목적

`training_loss`는 표본마다 t를 균등하게 선택하고 노이즈 ε와 모델 출력 사이의 모든 원소 MSE를 반환한다. 시간별 가중치나 관측 우도 항을 계산하지 않는다. 파라미터화가 연결되어 있어도 이 단순 손실이 정확한 전체 ELBO와 수치적으로 같은 것은 아니다.

[DDPM 논문](https://arxiv.org/abs/2006.11239)은 디노이징 스코어 매칭과 가중 변분 목적의 연결을 다룬다. 출력의 'DSM=노이즈 예측 손실'은 가우시안 전이와 시간별 배율·가중치 조건을 붙여 해석해야 한다. 스코어 회귀와 노이즈 회귀는 배율 변환에 따라 시간별 오차의 가중치가 달라진다.

## 4. 역평균과 분산 선택

`p_sample`은 모델의 노이즈 예측으로 다음 평균을 계산한다.

$$\mu_\theta(x_t,t)=\frac1{\sqrt{\alpha_t}}
\left(x_t-\frac{\beta_t}{\sqrt{1-\bar\alpha_t}}\epsilon_\theta(x_t,t)\right)$$

추가 잡음 분산은 βt를 사용하며 마지막 인덱스 0에서는 잡음을 생략한다. 이는 조건부 전방향 사후분산 $\widetilde\beta_t=\beta_t(1-\bar\alpha_{t-1})/(1-\bar\alpha_t)$를 사용하는 선택과 구분한다. 스케줄과 분산·최종 관측 처리의 관례가 모델을 정의한다.

잡음 생략 여부를 **t[0] 하나로 결정**하므로 p_sample은 같은 단계의 배치라는 전제를 갖는다. sample 메서드는 그 전제를 만족하지만, 호출자가 t=0과 t>0을 섞으면 표본별 처리가 맞지 않는다. β가 0~1 사이이고 단계 수가 양수인지 등의 입력 검증도 이 클래스에 없다.

## 5. 장치·모드·기울기 기록

스케줄과 초기 샘플, 시간 텐서는 CPU에 생성한다. DDPM은 nn.Module이 아니므로 모델에 .to(device)를 호출한다고 스케줄이 함께 이동하지 않는다. CUDA나 다른 dtype으로 쓰려면 관련 텐서와 조건을 일치시키는 별도 처리가 필요하다.

sample과 p_sample에는 no_grad나 eval 전환이 없다. 학습 가능한 모델을 그대로 넣으면 모든 생성 단계의 계산 그래프가 쌓여 메모리 사용이 커질 수 있다. Dropout·BatchNorm을 포함한 모델의 추론 모드도 호출자가 정해야 한다. 클래스 정의만으로 완성된 학습·추론 파이프라인을 갖췄다고 볼 수 없다.

| VP SDE와 DDPM | 연속시간 극한과 가우시안 노이즈 구조 | 유한 보폭의 전이와 수치 오차 |
| 확률 흐름 ODE와 DDIM | 결정적인 생성 경로와 연속시간 연결 | 모든 유한 단계가 같은 적분법이라는 주장 |

[DDIM 논문](https://arxiv.org/abs/2010.02502)은 같은 학습 절차에서 다른 생성 전이를 구성하는 방법을 다룬다. DDIM 계열에는 확률적인 설정도 있으므로 DDIM이라는 이름만으로 항상 결정적이라고 부르지 않는다. 초기값을 고정한 결정적 전이도 잠재 초기화 자체의 무작위성은 남는다.

출력에 나열된 가이던스, 잠재 확산, 영상·분자 생성 등은 확장 방향의 예시다. 이 DDPM 클래스가 구현하거나 성능을 비교한 기능은 아니다. 특정 모델 이름만으로 계산비용이나 품질의 우열을 일반화하지 않는다.

## 연습문제

<div class="drillbox" markdown>

**연습문제 1.** <span class="diff med" title="중간"></span>
B×C×H×W 입력에서 q_sample의 스케줄 계수는 어떤 모양으로 확장되는가?

</div>

??? success "연습문제 1 풀이"

    길이 B의 t로 고른 계수는 B이고 unsqueeze를 반복해 B×1×1×1로 만든다. 입력·노이즈와 브로드캐스팅해 같은 B×C×H×W 관측을 만든다. 스케줄·인덱스·입력의 장치와 범위가 맞아야 한다.



---

<div class="drillbox" markdown>

**연습문제 2.** <span class="diff hard" title="어려움"></span>
노이즈 예측 MSE를 스코어 제곱오차로 바꿀 때 배율을 설명하라.

</div>

??? success "연습문제 2 풀이"

    주변 스코어는 노이즈 조건부 평균에 −1/√(1−ᾱt)를 곱한 값이다. 같은 배율로 예측을 변환하면 스코어 오차의 제곱은 노이즈 예측 오차 제곱의 1/(1−ᾱt)배다. 따라서 시간별 가중치를 무시하고 두 목적의 수치가 같다고 할 수 없다.



---

<div class="drillbox" markdown>

**연습문제 3.** <span class="diff med" title="중간"></span>
p_sample에 서로 다른 시간의 배치를 전달하거나 sample을 학습 모드에서 부르면 어떤 문제가 생길 수 있는가?

</div>

??? success "연습문제 3 풀이"

    잡음 생략 여부가 t[0]에만 따라 서로 다른 시간의 배치에서 잘못 적용될 수 있다. sample은 eval·no_grad를 설정하지 않아 모델 모드에 따른 출력 차이와 긴 계산 그래프가 남을 수 있다. 이 스크립트는 클래스만 정의하므로 실제 학습·품질·스케줄 비교 결과도 없다.



## 정리하며

스코어 모델과 DDPM은 가우시안 노이즈의 배율 관계로 연결되지만 손실 가중치와 샘플러는 구분해야 한다. 이 클래스는 CPU 스케줄, 동일 시간 배치와 외부 추론 설정을 전제로 한다. 설명 출력과 실제 학습·생성의 검증을 혼동하지 않는다.
